cmake_minimum_required(VERSION 3.22)
project(EnsembleCue VERSION 0.2.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
  add_compile_options(/utf-8)
endif()
option(ENSEMBLECUE_GUI "Build Windows standalone application" ON)
add_library(band_core src/Core.cpp)
target_include_directories(band_core PUBLIC src)
if(MSVC)
  target_compile_options(band_core PRIVATE /W4 /utf-8)
endif()
enable_testing()
add_executable(core_tests tests/CoreTests.cpp)
target_link_libraries(core_tests PRIVATE band_core)
add_test(NAME core_tests COMMAND core_tests)
add_executable(sample_rate_tests tests/SampleRates.cpp)
target_link_libraries(sample_rate_tests PRIVATE band_core)
add_test(NAME sample_rate_tests COMMAND sample_rate_tests)
add_executable(ensemble_tests tests/EnsembleTests.cpp)
target_link_libraries(ensemble_tests PRIVATE band_core)
add_test(NAME ensemble_tests COMMAND ensemble_tests)
add_executable(benchmark tests/Benchmark.cpp)
target_link_libraries(benchmark PRIVATE band_core)
add_executable(quality_benchmark tests/Quality.cpp)
target_link_libraries(quality_benchmark PRIVATE band_core)
if(ENSEMBLECUE_GUI)
  add_subdirectory(third_party/JUCE EXCLUDE_FROM_ALL)
  juce_add_gui_app(EnsembleCue PRODUCT_NAME "EnsembleCue" COMPANY_NAME "EnsembleCue")
  target_sources(EnsembleCue PRIVATE src/Main.cpp src/Engine.cpp)
  target_link_libraries(EnsembleCue PRIVATE band_core juce::juce_audio_utils juce::juce_gui_extra)
  target_compile_definitions(EnsembleCue PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_ASIO=1 JUCE_USE_FLAC=0 JUCE_USE_OGGVORBIS=0 JUCE_MODAL_LOOPS_PERMITTED=1)
  if(MSVC)
    target_compile_options(EnsembleCue PRIVATE /utf-8)
  endif()
  juce_add_console_app(drum_render PRODUCT_NAME "drum_render")
  target_sources(drum_render PRIVATE src/Render.cpp)
  target_link_libraries(drum_render PRIVATE band_core juce::juce_audio_formats)
  target_compile_definitions(drum_render PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_USE_FLAC=0 JUCE_USE_OGGVORBIS=0)
  juce_add_console_app(audio_probe PRODUCT_NAME "audio_probe")
  target_sources(audio_probe PRIVATE tests/AudioProbe.cpp src/Engine.cpp)
  target_link_libraries(audio_probe PRIVATE band_core juce::juce_audio_utils juce::juce_gui_extra psapi)
  target_compile_definitions(audio_probe PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_ASIO=1 JUCE_USE_FLAC=0 JUCE_USE_OGGVORBIS=0 JUCE_MODAL_LOOPS_PERMITTED=1)
endif()
